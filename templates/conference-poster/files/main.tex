% ---------------------------------------------------------------- heat raster
% A classed thermal grid drawn from a smooth field; it echoes a Landsat tile.
% \lsheatraster{columns}{rows}{cell size}{background colour}{coldest class colour}
\newcommand{\lsheatraster}[5]{%
  \pgfmathsetmacro\lscols{#1-1}\pgfmathsetmacro\lsrows{#2-1}%
  \foreach \i in {0,...,\lscols}{\foreach \j in {0,...,\lsrows}{%
    \pgfmathsetmacro\u{\i/\lscols}\pgfmathsetmacro\v{\j/\lsrows}%
    \pgfmathsetmacro\heat{exp(-((\u-0.38)^2+(\v-0.6)^2)/0.035)
      +0.75*exp(-((\u-0.74)^2+(\v-0.28)^2)/0.02)
      -0.7*exp(-((\v-0.15-0.7*\u)^2)/0.004)
      +0.09*sin(37*\u r)*cos(23*\v r)+0.06}%
    \pgfmathtruncatemacro\heatclass{20*round(5*max(0,min(1,\heat)))}%
    \ifnum\heatclass=0 \def\lscell{#5}\else\def\lscell{lsAccent!\heatclass!#4}\fi
    \fill[\lscell] (\i*#3,\j*#3) rectangle ++(0.84*#3,0.84*#3);
  }}%
}
