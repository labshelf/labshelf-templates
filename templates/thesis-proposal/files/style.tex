% style.tex -- the whole visual design of the proposal lives here.

% ---- Fonts --------------------------------------------------------------
% Alegreya for text (old-style figures sit well in running prose),
% Alegreya Sans for headings, labels and tables.
\usepackage[T1]{fontenc}
\usepackage[osf]{Alegreya}
\usepackage[osf]{AlegreyaSans}
\usepackage[final]{microtype}

% ---- Page geometry ------------------------------------------------------
% A ~70-character measure with generous margins reads comfortably on A4.
\usepackage[a4paper, textwidth=142mm, textheight=232mm,
            hmarginratio=1:1, vmarginratio=4:5, headsep=9mm,
            footskip=12mm]{geometry}
\linespread{1.08}
\setlength{\parindent}{1.2em}

% ---- Colour -------------------------------------------------------------
% One warm terracotta accent, a darker shade for links and a pale tint
% for panels. Change these three lines to recolour the document.
\usepackage{xcolor}
\definecolor{accent}{HTML}{B4532A}
\definecolor{accentdark}{HTML}{8A3A1C}
\definecolor{accenttint}{HTML}{F8EEE7}
\definecolor{ink}{HTML}{2B2522}

% ---- Headings -----------------------------------------------------------
% Chapters are section-style (no page break) so a short proposal flows;
% the number hangs in the margin in terracotta, over a hairline.
\RedeclareSectionCommand[style=section, beforeskip=-3.2\baselineskip,
  afterskip=1.2\baselineskip, font=\sffamily\Large\color{ink}]{chapter}
\RedeclareSectionCommand[beforeskip=-1.6\baselineskip,
  afterskip=.5\baselineskip, font=\sffamily\large\color{ink}]{section}
\RedeclareSectionCommand[beforeskip=-1.2\baselineskip,
  afterskip=.3\baselineskip,
  font=\sffamily\normalsize\itshape\color{accentdark}]{subsection}
\setkomafont{disposition}{\sffamily}
\renewcommand*{\chapterformat}{%
  \makebox[0pt][r]{\color{accent}\thechapter\hspace{.7em}}}
\renewcommand*{\sectionformat}{%
  \makebox[0pt][r]{\color{accent}\thesection\hspace{.6em}}}
\renewcommand*{\subsectionformat}{}
\makeatletter
\renewcommand*{\sectionlinesformat}[4]{%
  \Ifstr{#1}{chapter}
    {\parbox{\linewidth}{\raggedright#3#4\par
       \vspace{.45\baselineskip}%
       {\color{accent}\rule{\linewidth}{.4pt}}}}
    {\@hangfrom{\hskip #2#3}{#4}}}
\makeatother

% ---- Research questions -------------------------------------------------
% \begin{researchquestion}{Short label} ... \end{researchquestion}
% numbers itself RQ1, RQ2, ... and can be referenced with \label/\ref.
\newcounter{rq}
\renewcommand*{\therq}{RQ\arabic{rq}}
\newtcolorbox{rqbox}{enhanced, breakable, colback=white, colframe=white,
  boxrule=0pt, arc=0pt, left=4.5mm, right=0mm, top=1mm, bottom=1mm,
  before skip=1em, after skip=1em, borderline west={.8pt}{0pt}{accent}}
\newenvironment{researchquestion}[1]{%
  \refstepcounter{rq}%
  \begin{rqbox}%
  {\sffamily\small\color{accent}\textsc{\MakeLowercase{\therq}}\enspace
   \color{ink}\bfseries #1}\par\vspace{.25em}\itshape\ignorespaces}
  {\end{rqbox}}
